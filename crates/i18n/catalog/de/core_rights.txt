# Rights that GitHub Projects need, and how to get them (setup, sync notes, failed actions)
core_right.missing = Dem GitHub-Zugang fehlt ein Recht: {what} geht nicht. {fix}
core_right.what_projects_read = GitHub Projects lesen
core_right.what_projects_write = GitHub Projects ändern
core_right.what_issues_read = Issues lesen
core_right.what_issues_write = Issues ändern
core_right.fix_classic = Gib dem klassischen Token den Bereich `{scope}` dazu, oder erstelle einen neuen Token damit
core_right.fix_fine_projects = Gib dem Fine-grained Token die Berechtigung „Projects“ der Organisation oder des Kontos (Zugriff: {level})
core_right.fix_fine_issues = Gib dem Fine-grained Token die Repository-Berechtigung „Issues“ (Zugriff: {level}) für die benutzten Repos
core_right.fix_app_projects = Gib der GitHub App die Berechtigung „Projects“ (Zugriff: {level}) und nimm die neue Berechtigung bei der Installation an
core_right.fix_app_issues = Gib der GitHub App die Repository-Berechtigung „Issues“ (Zugriff: {level}) und nimm die neue Berechtigung bei der Installation an
core_right.fix_unknown = Klassischer Token: Bereich `{scope}`. Fine-grained Token oder GitHub App: Berechtigung „{permission}“ (Zugriff: {level})
