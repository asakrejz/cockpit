# The "Translate" button for texts from Jira and GitHub
translate.button = Translate to {language}
translate.busy = Translating …
translate.done = Translated by the model. Check important details in the original.
translate.show_original = Show original
translate.no_key = Needs a key and a model for the agent (see Set up).
translate.notice = The text is sent to the model service you set up for the agent (through the local server) and translated there. Only this one text is sent, and only now.
translate.confirm = Translate
translate.cancel = Cancel
